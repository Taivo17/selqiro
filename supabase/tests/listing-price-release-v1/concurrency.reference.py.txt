# Integration driver methods copied into the guarded external runner.
# Not standalone; uses that runner's pinned new helper, bounded commands and cleanup.
    def start_session(self, name, sql):
        self.label_guard(name); self.helper_check(True)
        op=self.out/'logs'/(name+'.stdout');ep=self.out/'logs'/(name+'.stderr')
        require(not op.exists() and not ep.exists(),'Paralleelse seansi logi juba olemas.')
        o=op.open('xb');e=ep.open('xb')
        args=self.docker('exec','-i','--user','postgres',self.helper,'env','-i',
          'PATH=/usr/lib/postgresql/bin:/usr/bin:/bin','HOME=/tmp','LC_ALL=C',
          'PGCLIENTENCODING=UTF8','PGPASSFILE=/tmp/no-price-passfile','PGCONNECT_TIMEOUT=4',
          'PGAPPNAME='+name,*self.psql_args())
        try:proc=subprocess.Popen(args,stdin=subprocess.PIPE,stdout=o,stderr=e,
                cwd=self.root,env=self.env,start_new_session=True)
        except BaseException:o.close();e.close();raise
        s=(proc,o,e,op,ep);self.sessions.append(s)
        proc.stdin.write(sql.encode());proc.stdin.flush();return s

    def end_session(self,s,final=''):
        proc,o,e,op,ep=s
        try:
            if final and proc.poll() is None:proc.stdin.write(final.encode());proc.stdin.flush()
            if not proc.stdin.closed:proc.stdin.close()
            proc.wait(timeout=12)
        finally:
            if proc.poll() is None:os.killpg(proc.pid,signal.SIGKILL);proc.wait()
            o.close();e.close()
        require(op.stat().st_size<=1024*1024 and ep.stat().st_size<=1024*1024,'Paralleelse seansi valjund liiga suur.')
        self.commands.append(dict(label=op.stem,exit_code=proc.returncode,async_session=True,finished_at=utc()))
        if s in self.sessions:self.sessions.remove(s)
        return proc.returncode,op.read_bytes(),ep.read_bytes()

    def race(self,name,a_sql,b_sql,expected_state,expected_message):
        a=b=None
        prefix="\\set ON_ERROR_STOP on\n\\set VERBOSITY verbose\nBEGIN;\nSET LOCAL statement_timeout='8s'; SET LOCAL idle_in_transaction_session_timeout='10s';\n"
        try:
            a=self.start_session('release-'+name+'-a',prefix+a_sql+'\n\\echo RELEASE_A_HELD\n')
            deadline=time.monotonic()+5
            while b'RELEASE_A_HELD' not in a[3].read_bytes().splitlines():
                require(a[0].poll() is None and time.monotonic()<deadline,'Esimene seanss ei hoidnud kontrollpunkti.')
                require(a[3].stat().st_size<1024*1024 and a[4].stat().st_size<1024*1024,'Seansi logi uletas piiri.')
                time.sleep(.02)
            b=self.start_session('release-'+name+'-b',prefix+b_sql+'\nCOMMIT;\n')
            waiting=False;deadline=time.monotonic()+4
            for n in range(20):
                if time.monotonic()>deadline or b[0].poll() is not None:break
                cp=self.sql("SELECT EXISTS(SELECT 1 FROM pg_stat_activity WHERE application_name='release-"+name+"-b' AND wait_event_type='Lock');",'race-'+name+'-wait-'+str(n))
                if cp.stdout.strip()==b't':waiting=True;break
                time.sleep(.02)
            ar=self.end_session(a,'COMMIT;\n');a=None;br=self.end_session(b);b=None
            require(waiting,'Tegelikku kattuvat lukku ei nahtud; katse ei ole PASS.')
            errors=re.findall(r'ERROR:\s+([0-9A-Z]{5}):\s+([^\r\n]+)',br[2].decode())
            require(ar[0]==0 and br[0]==3 and len(errors)==1 and errors[0]==(expected_state,expected_message),
                'Kahe uhenduse vea tulemus erineb: '+name)
            return dict(name=name,overlapping_lock_observed=True,winner_exit=ar[0],loser_exit=br[0],
                sqlstate=expected_state,message=expected_message)
        finally:
            for s in (a,b):
                if s is not None:
                    try:self.end_session(s,'ROLLBACK;\n')
                    except Exception:pass

    def concurrency_tests(self):

        v='44444444-4444-4444-8444-444444444444';k='cccccccc-cccc-4ccc-8ccc-cccccccccccc'
