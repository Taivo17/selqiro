const {test}=require('node:test'),assert=require('node:assert/strict');
const fs=require('node:fs'),path=require('node:path'),vm=require('node:vm');
const base=path.resolve(__dirname,'..'),ts=require(process.env.SELQIRO_TYPESCRIPT_FILE||'typescript');
const clone=x=>JSON.parse(JSON.stringify(x)),cache=new Map();
function mod(file){file=path.resolve(file);assert.ok(file.startsWith(base+path.sep));if(cache.has(file))return cache.get(file).exports;
 const code=ts.transpileModule(fs.readFileSync(file,'utf8'),{fileName:file,reportDiagnostics:true,compilerOptions:{module:ts.ModuleKind.CommonJS,target:ts.ScriptTarget.ES2017,strict:true}});
 assert.equal(code.diagnostics.filter(x=>x.category===ts.DiagnosticCategory.Error).length,0);const m={exports:{}};cache.set(file,m);
 vm.runInNewContext(code.outputText,{module:m,exports:m.exports,URL,Date,BigInt,Number,Object,Set,
 require:name=>{assert.ok(name.startsWith('.'));return mod(path.resolve(path.dirname(file),name+'.ts'));}},{filename:file,timeout:2000});return m.exports;}

 title:'surface-alpha',description:'72 in and 5 km remain authored text',title_truncated:false,description_truncated:false,price:fixed(),
 category:'vehicles',subcategory:'cars',detail_category:'passenger_cars',condition:'used',country:'Eesti',city:'Paide',
 image_url:'https://example.invalid/primary.jpg',images:[image()],images_has_more:false,
 details:[{key:'vin',label:'VIN',value:'PUBLIC-AUTHORED-VIN'}],details_truncated:false,
 seller_name:'Seller A',seller_slug:'seller-a',seller_avatar_url:null,seller_type:'private',created_at:'2026-01-01T00:00:00+00:00',active_until:'2099-01-01T00:00:00+00:00'}};}

 category:d.category,subcategory:d.subcategory,detail_category:d.detail_category,condition:d.condition,country:d.country,city:d.city,
 image_url:d.image_url,seller_name:d.seller_name,seller_slug:d.seller_slug,seller_avatar_url:d.seller_avatar_url,seller_type:d.seller_type,created_at:d.created_at};}
function profile(total='1',offset=0,limit=12){const n=BigInt(total),remain=n>BigInt(offset)?n-BigInt(offset):0n,more=n>BigInt(offset+limit),cap=more&&offset+limit>100000;
 return {schema_version:1,content_scope:'public_profile_ordinary_listings',seller_slug:'seller-a',profile_available:true,store_category_id:null,sort:'newest',
 items:Array.from({length:Number(remain>BigInt(limit)?BigInt(limit):remain)},(_,i)=>card(String(1001+offset+i))),total_count:total,result_limit:limit,result_offset:offset,
 has_more:more,next_offset:more&&!cap?offset+limit:null,window_limit_reached:cap};}
