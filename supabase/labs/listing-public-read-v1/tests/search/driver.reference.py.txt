    def start_session(self,name,sql):
        self.label_guard(name);require(len(sql.encode())<=65536,'Sunteetiline session liiga suur.')
        self.helper_check(True)
        target=self.out/'logs';op=target/(name+'.stdout');ep=target/(name+'.stderr')
        require(not op.exists() and not ep.exists(),'Race session ei kordu.')
        o=op.open('xb');e=ep.open('xb')
        args=self.docker('exec','-i','--user','postgres',self.helper,'env','-i',
          'PATH=/usr/lib/postgresql/bin:/usr/bin:/bin','HOME=/tmp','LC_ALL=C','PGCLIENTENCODING=UTF8',
          'PGPASSFILE=/tmp/no-price-passfile','PGCONNECT_TIMEOUT=4','PGAPPNAME='+name,*self.psql_args())
        if not hasattr(self,'session_records'):self.session_records={}
        rec=dict(name=name,started_at=now(),finished_at=None,held_observed_at=None,submitted='',exit_code=None)
        self.session_records[name]=rec
        proc=subprocess.Popen(args,stdin=subprocess.PIPE,stdout=o,stderr=e,cwd=self.root,env=self.env,start_new_session=True)
        self.children.append(proc);os.set_blocking(proc.stdin.fileno(),False);s=(proc,o,e,op,ep)
        self.put('tests/concurrency/'+name+'.initial.sql',sql)
        try:self._session_write(s,sql.encode())
        except BaseException:
            if proc.poll() is None:os.killpg(proc.pid,signal.SIGKILL);proc.wait()
            o.close();e.close();raise
        return s

    def end_session(self,s,final=''):
        proc,o,e,op,ep=s;rec=self.session_records[op.stem]
        try:
            if final:self._session_write(s,final.encode())
            proc.stdin.close();proc.wait(timeout=12)
        finally:
            if proc.poll() is None:os.killpg(proc.pid,signal.SIGKILL);proc.wait()
            o.close();e.close();rec['finished_at']=now();rec['exit_code']=proc.returncode
            path='tests/concurrency/'+op.stem+'.submitted.sql';self.put(path,rec['submitted'])
            published={k:v for k,v in rec.items() if k!='submitted'}
            published.update(sql_path=path,sql_sha256=sha(rec['submitted'].encode()),
              stdout_path='logs/'+op.name,stderr_path='logs/'+ep.name,
              stdout_sha256=sha(op.read_bytes()),stderr_sha256=sha(ep.read_bytes()))
            self.js('tests/concurrency/'+op.stem+'.json',published)
            self.commands.append(dict(label=op.stem,exit_code=proc.returncode,async_session=True,
              started_at=rec['started_at'],finished_at=rec['finished_at'],sql_sha256=published['sql_sha256']))
        require(op.stat().st_size<=1024*1024 and ep.stat().st_size<=1024*1024,'Race valjund liiga suur.')
        return proc.returncode,op.read_bytes(),ep.read_bytes()

    def wait_ready(self,s):
        proc,_,_,op,ep=s;start=time.monotonic()
        while time.monotonic()-start<5:
            require(proc.poll() is None,'Race A loppes enne kontrollpunkti.')
            require(op.stat().st_size<=1024*1024 and ep.stat().st_size<=1024*1024,'Race valjund uletas piiri.')
            if b'PRICE_A_HELD' in op.read_bytes().splitlines():return
            time.sleep(.02)
        raise Stop('Race A kontrollpunkti ei saabunud; ei nimeta katset labinuks.')

    def _session_write(self,s,data):
        proc,o,e,op,ep=s
        pending=memoryview(data);start=time.monotonic()
        while pending:
            require(time.monotonic()-start<4 and proc.poll() is None,'Sessioni sisend ei saanud piiratud ajaga edastatud.')
            require(op.stat().st_size<=1024*1024 and ep.stat().st_size<=1024*1024,'Sessioni valjund uletas piiri.')
            try:pending=pending[os.write(proc.stdin.fileno(),pending[:65536]):]
            except BlockingIOError:time.sleep(.01)
        self.session_records[op.stem]['submitted']+=data.decode()

    def test_batch(self):
        parts=[('batch/preamble',"\\set ON_ERROR_STOP on\nBEGIN;\nSET LOCAL TimeZone='UTC';\nSET LOCAL statement_timeout='15s';\n")]
        for n in ['01_value_contract.sql','02_write_guard.sql','03_owner_price_core.sql','04_basics_contract.sql','05_owner_basics_atomic.sql']:
            parts.append(('dependencies/'+n,self.payload['dependencies/'+n].decode()+'\n'))
        for n in ['10_price_read_model.sql','11_public_search_v2.sql']:parts.append(('candidate/'+n,self.payload['candidate/'+n].decode()+'\n'))
        for n in ['reader-seed.sql','reader-suite.sql']:parts.append(('tests/'+n,self.payload['tests/'+n].decode()+'\n'))
        parts.append(('batch/rollback','ROLLBACK;\n'));line=1;self.batch_map=[]
        for name,t in parts:
            self.batch_map.append(dict(path=name,start_line=line,end_line=line+t.count('\n')-1,bytes=len(t.encode()),sha256=sha(t.encode())))
            line+=t.count('\n')
        self.js('tests/executed-source-map.json',self.batch_map)
        return ''.join(t for _,t in parts)

    def snapshot_tests(self):
        results=[]
        for i,final in enumerate(['COMMIT;\n','ROLLBACK;\n']):
            self.sql("UPDATE public.listings SET price_kind='fixed',price_amount=10,currency='EUR' WHERE id=1001;",'snapshot-reset-'+str(i))
            name='price-read-snapshot-'+str(i);a=b=None
            try:
                a=self.start_session(name+'-a',"\\set ON_ERROR_STOP on\nBEGIN;\nSET LOCAL statement_timeout='6s'; SET LOCAL idle_in_transaction_session_timeout='10s';\nUPDATE public.listings SET price_kind='fixed',price_amount=999999999999999999.999,currency='KWD' WHERE id=1001;\n\\echo PRICE_A_HELD\n")
                self.wait_ready(a);held=now()
                active="SELECT EXISTS(SELECT 1 FROM pg_stat_activity WHERE application_name='"+name+"-a' AND state='idle in transaction');"
                require(self.sql(active,'writer-held-before-'+str(i)).stdout.strip()==b't','A tehingu hoidmist ei kinnitatud.')
                b=self.start_session(name+'-b',"\\set ON_ERROR_STOP on\nBEGIN READ ONLY;\nSET LOCAL statement_timeout='3s';\nSET LOCAL ROLE anon;\nSELECT public.search_public_listings_v2(p_search_query=>'reader-alpha');\nCOMMIT;\n")
                br=self.end_session(b);b=None;read_finished=now()
                require(br[0]==0,'B lugemine ei loppenud enne A vabastamist.')
                pages=[load(x) for x in br[1].splitlines() if x.startswith(b'{')]
                require(len(pages)==1 and pages[0]['items'][0]['price']['amount']=='10' and pages[0]['items'][0]['price']['currency']=='EUR','B ei lugenud tervet vana committed hinnaseisu.')
                require(self.sql(active,'writer-held-after-'+str(i)).stdout.strip()==b't','A ei hoidnud tehingut B lugemise lopus.')
                release=now();ar=self.end_session(a,final);a=None;require(ar[0]==0,'A tehingu lopp ei labinud.')
                q="BEGIN READ ONLY; SET LOCAL ROLE anon; SELECT public.search_public_listings_v2(p_search_query=>'reader-alpha'); COMMIT;"
                cp=self.sql(q,'reader-after-'+str(i));post=load(cp.stdout.strip())
                expected=('999999999999999999.999','KWD') if i==0 else ('10','EUR')
                v=post['items'][0]['price'];require((v['amount'],v['currency'])==expected,'Vabastatud tehingu jarel hinnaseis erineb.')
                rec=dict(name=name,passed=True,read_only_transaction=True,writer_held_at=held,reader_finished_at=read_finished,
                  writer_released_at=release,first_final=final.strip(),old_amount='10',old_currency='EUR',after_amount=v['amount'],after_currency=v['currency'],
                  property='B completes while A remains idle-in-transaction; complete old price then complete committed price, no dirty/mixed tuple.')
                results.append(rec);self.js('concurrent-read-summary.json',dict(passed=len(results)==2,tests=results,expected_count=2))
            finally:

                        try:self.end_session(session,'ROLLBACK;\n' if session[0].poll() is None else '')
                        except Exception:pass
        self.result['CONCURRENT_READ']='PASS_TWO_NONBLOCKING_SNAPSHOT_CASES'
