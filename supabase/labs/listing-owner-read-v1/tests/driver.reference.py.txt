    def test_batch(self):

        for n in ['dependencies/01_value_contract.sql','dependencies/02_write_guard.sql','dependencies/10_price_read_model.sql',
          'candidate/15_owner_money_models.sql','candidate/16_owner_marketplace_read_v3.sql','dependencies/seed.sql','tests/owner-seed.sql','tests/owner-suite.sql']:
            parts.append((n,self.payload[n].decode()+'\n'))
        parts.append(('batch/rollback','ROLLBACK;\n'));line=1;self.batch_map=[]
        for name,t in parts:
            self.batch_map.append(dict(path=name,start_line=line,end_line=line+t.count('\n')-1,bytes=len(t.encode()),sha256=sha(t.encode())))
            line+=t.count('\n')
        self.js('tests/executed-source-map.json',self.batch_map)
        return ''.join(t for _,t in parts)
